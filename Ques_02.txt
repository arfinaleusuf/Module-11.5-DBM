CREATE TABLE Class (
    ClassID INT PRIMARY KEY,
    ClassName VARCHAR(50),
    Section VARCHAR(10)
);

CREATE TABLE Student (
    StudentID INT PRIMARY KEY,
    Name VARCHAR(100),
    RollNo INT UNIQUE,
    Gender VARCHAR(10),
    ClassID INT,
    FOREIGN KEY (ClassID)
        REFERENCES Class(ClassID)
);

CREATE TABLE Teacher(
    TeacherID INT PRIMARY KEY,
    TeacherName VARCHAR(50),
    Subject VARCHAR(20)
);
